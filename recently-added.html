    Use in RStudio

    Access live Butterfly Explorers data directly in R using the rbutterflyexplorers package.

    1 Install the package 
    # From r-universe
install.packages("rbutterflyexplorers",
  repos = "https://sajankc143.r-universe.dev")

# Or from GitHub
devtools::install_github("sajankc143/rbutterflyexplorers")
    2 Fetch observations 
    library(rbutterflyexplorers)

obs <- be_observations()                          # all records
obs <- be_observations(species = "Agathymus")     # by species
obs <- be_observations(country = "Costa Rica")    # by country
obs <- be_observations(state_province = "Florida")# by state
obs <- be_observations(date_from = "2023-01-01")  # by date
spp <- be_species()                               # species list
    3 Map your data 
    library(sf); library(ggplot2)

obs |>
  dplyr::filter(!is.na(decimalLatitude)) |>
  sf::st_as_sf(coords = c("decimalLongitude",
                           "decimalLatitude"), crs = 4326) |>
  ggplot() + geom_sf(aes(color = family))
